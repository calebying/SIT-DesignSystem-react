# Radio

same pattern used for `Checkbox` and `Form/Switch.tsx`.

## Radio.Group

Groups radios under one shared `name` (native mutual exclusivity) plus a controlled
`value`/`onChange` pair.

export const GroupTemplate = () => {
  const [value, setValue] = useState('email');
  return (
    <Radio.Group label="Preferred contact method" value={value} onChange={setValue}>
      <Radio id="contact-email" value="email" label="Email" />
      <Radio id="contact-phone" value="phone" label="Phone" />
      <Radio id="contact-mail" value="mail" label="Mail" />
    </Radio.Group>
  );
};
